// 교육유형 버튼 컨테이너만 복원 (만족도 제외) modal.querySelectorAll('[data-react-hidden="true"]').forEach(function(el) { // 만족도 관련 요소는 복원하지 않음 if (el.textContent.includes('만족도')) return; el.removeAttribute('data-react-hidden'); el.style.removeProperty('display'); }); // 인원/시간 input 자유 숫자 입력 modal.querySelectorAll('input[type="number"]').forEach(function(inp) { if (inp.getAttribute('data-free-input') === 'true') return; inp.removeAttribute('min'); inp.removeAttribute('max'); inp.removeAttribute('step'); inp.setAttribute('data-free-input', 'true'); }); modal.querySelectorAll('input').forEach(function(inp) { if (inp.getAttribute('data-free-input') === 'true') return; var label = modal.querySelector('label[for="' + inp.id + '"]'); var labelText = label ? label.textContent : ''; var prevText = inp.previousElementSibling ? inp.previousElementSibling.textContent : ''; if (labelText.includes('인원') || labelText.includes('시간') || prevText.includes('인원') || prevText.includes('시간')) { inp.removeAttribute('min'); inp.removeAttribute('max'); inp.removeAttribute('step'); inp.setAttribute('type', 'number'); inp.setAttribute('data-free-input', 'true'); } }); // 만족도 입력란 숨기기 (placeholder로 정확히 찾기) modal.querySelectorAll('input').forEach(function(inp) { if (inp.placeholder && inp.placeholder.includes('0.0')) { var container = inp.parentElement; while (container && container !== modal) { if (container.parentElement === modal) break; container = container.parentElement; } if (container && container !== modal) { container.style.setProperty('display', 'none', 'important'); } } }); // 센터 직접 입력 기능 추가 addCenterDirectInput(modal); } function addCenterDirectInput(modal) { // 전체 문서 기준으로 중복 체크 (modal이 매번 다른 div일 수 있음) if (document.querySelector('[data-center-input]')) return; // 센터 select 찾기 (옵션 5개 이상, 방식/강사 아닌 것) var centerSel = null; modal.querySelectorAll('select').forEach(function(sel) { if (sel.id === 'my-instr-select') return; var opts = Array.from(sel.options); var texts = opts.map(function(o) { return o.text; }); if (opts.length >= 5 && !texts.some(function(t) { return t === '온라인' || t === '오프라인'; }) && !texts.some(function(t) { return ['김보경','김현정','나혜리','구혜린'].indexOf(t) > -1; })) { centerSel = sel; } }); if (!centerSel) return; // localStorage에 저장된 커스텀 센터 불러오기 var customCenters = JSON.parse(localStorage.getItem('metamEdu_customCenters') || '[]'); customCenters.forEach(function(val) { var exists = Array.from(centerSel.options).some(function(o) { return o.value === val; }); if (!exists) { var opt = document.createElement('option'); opt.value = val; opt.text = val; centerSel.appendChild(opt); } }); // 직접 입력 UI 생성 var wrapper = document.createElement('div'); wrapper.setAttribute('data-center-input', 'true'); wrapper.style.cssText = 'display:flex;gap:6px;align-items:center;margin-top:6px;'; var input = document.createElement('input'); input.type = 'text'; input.placeholder = '새 센터명 직접 입력'; input.style.cssText = 'flex:1;padding:8px 10px;border:1px solid #e2e8f0;border-radius:8px;font-size:13px;outline:none;'; var addBtn = document.createElement('button'); addBtn.type = 'button'; addBtn.textContent = '추가'; addBtn.style.cssText = 'padding:8px 14px;background:#3b82f6;color:white;border:none;border-radius:8px;font-size:13px;cursor:pointer;white-space:nowrap;'; addBtn.addEventListener('click', function() { var val = input.value.trim(); if (!val) return; // 드롭다운에 추가 후 선택 var exists = Array.from(centerSel.options).some(function(o) { return o.value === val; }); if (!exists) { var opt = document.createElement('option'); opt.value = val; opt.text = val; centerSel.appendChild(opt); } centerSel.value = val; // localStorage 저장 var saved = JSON.parse(localStorage.getItem('metamEdu_customCenters') || '[]'); if (!saved.includes(val)) { saved.push(val); localStorage.setItem('metamEdu_customCenters', JSON.stringify(saved)); } input.value = ''; addBtn.textContent = '✓ 추가됨'; setTimeout(function() { addBtn.textContent = '추가'; }, 1500); }); input.addEventListener('keydown', function(e) { if (e.key === 'Enter') { e.preventDefault(); addBtn.click(); } }); wrapper.appendChild(input); wrapper.appendChild(addBtn); centerSel.parentElement.appendChild(wrapper); }
